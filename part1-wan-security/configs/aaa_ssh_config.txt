# R3 LAN 1 (PC-B):              192.168.20.0/24
#   R3 G0/0:                    192.168.20.1
#   PC-B:                       192.168.20.10
#
# R3 LAN 2 (PC-C):              192.168.30.0/24
#   R3 G0/1:                    192.168.30.1
#   PC-C:                       192.168.30.10
#
# R1–R2 WAN:                    10.0.0.0/30
#   R1 S0/0/0:                  10.0.0.1
#   R2 S0/0/0:                  10.0.0.2
#
# R2–R3 WAN:                    10.0.0.4/30
#   R2 S0/0/1:                  10.0.0.5
#   R3 S0/0/1:                  10.0.0.6

============================================================
R1 CONFIG – ROUTER (PC-A SERVER LAN)
============================================================
enable
conf t
hostname R1

! Secure vs non-secure enable
enable secret CISCO123
enable password cisco

ip domain-name sait.local

! Local users (for console/VTY)
username admin secret Admin123
username sshuser secret Ssh12345

! LAN interface for PC-A/server
interface g0/1
 description LAN_PC-A_SERVER
 ip address 192.168.10.1 255.255.255.0
 no shutdown

! WAN towards R2
interface s0/0/0
 description LINK_TO_R2
 ip address 10.0.0.1 255.255.255.252
 clock rate 64000
 no shutdown

! RIP routing
router rip
 version 2
 no auto-summary
 network 192.168.10.0
 network 10.0.0.0
exit

! Console & VTY (local auth)
line console 0
 password cisco
 login
 logging synchronous
exit

line vty 0 4
 login local
 transport input telnet ssh
exit

! SSH (keys)
crypto key generate rsa
 1024
ip ssh version 2

end
write memory

============================================================
R3 CONFIG – ROUTER (PC-B, PC-C, AAA LOCAL)
============================================================
enable
conf t
hostname R3

enable secret CISCO123
enable password cisco

ip domain-name sait.local

! Local users (will be used by AAA)
username admin secret Admin123
username sshuser secret Ssh12345

! LAN for PC-B
interface g0/0
 description LAN_PC-B
 ip address 192.168.20.1 255.255.255.0
 no shutdown

! LAN for PC-C
interface g0/1
 description LAN_PC-C
 ip address 192.168.30.1 255.255.255.0
 no shutdown

! WAN from R2
interface s0/0/1
 description LINK_TO_R2
 ip address 10.0.0.6 255.255.255.252
 no shutdown

! RIP
router rip
 version 2
 no auto-summary
 network 192.168.20.0
 network 192.168.30.0
 network 10.0.0.0
exit

! ===== AAA LOCAL AUTHENTICATION =====
aaa new-model
aaa authentication login CONSOLE local
aaa authentication login VTY local

line console 0
 logging synchronous
 login authentication CONSOLE
exit

line vty 0 4
 login authentication VTY
 transport input telnet ssh
exit

! SSH
crypto key generate rsa
 1024
ip ssh version 2

end
write memory
